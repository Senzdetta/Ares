package main

import (
	"bufio"
	"embed"
	"flag"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"sync"
	"time"

	"github.com/hashicorp/hcl/v2/hclsimple"
)

//go:embed comment.hcl
var embeddedConfig embed.FS

// Struktur Mapping HCL
type ConfigFile struct {
	Languages []LanguageConfig `hcl:"language,block"`
}

type LanguageConfig struct {
	Name string   `hcl:"name,label"`
	Ext  []string `hcl:"ext"`
	Com  []string `hcl:"com"`
}

// Rule internal yang dipakai worker
type CommentRule struct {
	Single     string
	IsBlock    bool
	BlockStart string
	BlockEnd   string
}

type FileStats struct {
	Path     string
	Lines    int
	Code     int
	Comments int
	Blanks   int
}

func main() {
	commentTypeFlag := flag.String("comment-type", "", "Path ke file config HCL eksternal")
	excludeFlag := flag.String("exclude", "", "Daftar file/folder yang di-ignore, dipisah koma")
	flag.Parse()

	args := flag.Args()
	root := "."
	if len(args) > 0 {
		root = args[0]
	}

	// 1. Load Aturan dari HCL
	rules, err := loadCommentRules(*commentTypeFlag)
	if err != nil {
		fmt.Printf("Error loading config: %v\n", err)
		return
	}

	// 2. Load Exclude List
	excludeMap := make(map[string]bool)
	excludeMap[".git"] = true
	excludeMap["vendor"] = true
	excludeMap["node_modules"] = true

	if *excludeFlag != "" {
		for _, item := range strings.Split(*excludeFlag, ",") {
			excludeMap[strings.TrimSpace(item)] = true
		}
	}

	runSLOC(root, rules, excludeMap, 8)
}

// Parser file HCL (support Embedded FS & Disk)
func loadCommentRules(configPath string) (map[string][]CommentRule, error) {
	var content []byte
	var err error

	if configPath != "" {
		content, err = os.ReadFile(configPath)
		if err != nil {
			return nil, fmt.Errorf("gagal membaca file config %s: %v", configPath, err)
		}
	} else {
		content, err = embeddedConfig.ReadFile("comment.hcl")
		if err != nil {
			return nil, fmt.Errorf("gagal membaca embedded config: %v", err)
		}
	}

	var hclConfig ConfigFile
	// Decode HCL ke struct Go
	err = hclsimple.Decode("comment.hcl", content, nil, &hclConfig)
	if err != nil {
		return nil, fmt.Errorf("gagal parsing HCL: %v", err)
	}

	// Mapping dari Extensi -> Daftar Rules Komentar
	rulesMap := make(map[string][]CommentRule)

	for _, lang := range hclConfig.Languages {
		var parsedRules []CommentRule

		for _, pattern := range lang.Com {
			if idx := strings.Index(pattern, "{}"); idx != -1 {
				prefix := pattern[:idx]
				suffix := pattern[idx+2:]

				if strings.HasPrefix(prefix, "/*") || strings.HasPrefix(prefix, "<!--") {
					parsedRules = append(parsedRules, CommentRule{
						IsBlock:    true,
						BlockStart: prefix,
						BlockEnd:   suffix,
					})
				} else {
					parsedRules = append(parsedRules, CommentRule{
						Single:  prefix,
						IsBlock: false,
					})
				}
			}
		}

		// Daftarkan rule ke setiap ekstensi dalam bahasa tersebut
		for _, ext := range lang.Ext {
			rulesMap[ext] = parsedRules
		}
	}

	return rulesMap, nil
}

func runSLOC(root string, rules map[string][]CommentRule, excludeMap map[string]bool, numWorkers int) {
	start := time.Now()

	filesChan := make(chan string, 1000)
	resultsChan := make(chan FileStats, 1000)
	var wg sync.WaitGroup

	for i := 0; i < numWorkers; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for path := range filesChan {
				ext := filepath.Ext(path)
				ruleList, exists := rules[ext]
				if !exists {
					continue
				}
				stats, err := countFile(path, ruleList)
				if err == nil {
					resultsChan <- stats
				}
			}
		}()
	}

	go func() {
		wg.Wait()
		close(resultsChan)
	}()

	go func() {
		filepath.WalkDir(root, func(path string, d os.DirEntry, err error) error {
			if err != nil {
				return nil
			}
			name := d.Name()
			if excludeMap[name] || (d.IsDir() && strings.HasPrefix(name, ".")) {
				if d.IsDir() {
					return filepath.SkipDir
				}
				return nil
			}

			if !d.IsDir() {
				ext := filepath.Ext(name)
				if _, ok := rules[ext]; ok {
					filesChan <- path
				}
			}
			return nil
		})
		close(filesChan)
	}()

	var totalFiles, totalLines, totalCode, totalComments, totalBlanks int
	for res := range resultsChan {
		totalFiles++
		totalLines += res.Lines
		totalCode += res.Code
		totalComments += res.Comments
		totalBlanks += res.Blanks
	}

	duration := time.Since(start)

	fmt.Printf("\n========================================\n")
	fmt.Printf("         SLOC ANALYSIS REPORT           \n")
	fmt.Printf("========================================\n")
	fmt.Printf("Target Direktori : %s\n", root)
	fmt.Printf("Files Scanned    : %d\n", totalFiles)
	fmt.Printf("Total Lines      : %d\n", totalLines)
	fmt.Printf("Code Lines       : %d\n", totalCode)
	fmt.Printf("Comments         : %d\n", totalComments)
	fmt.Printf("Blank Lines      : %d\n", totalBlanks)
	fmt.Printf("Waktu Eksekusi   : %v\n", duration)
	fmt.Printf("========================================\n")
}

func countFile(path string, rules []CommentRule) (FileStats, error) {
	file, err := os.Open(path)
	if err != nil {
		return FileStats{}, err
	}
	defer file.Close()

	scanner := bufio.NewScanner(file)
	var lines, code, comments, blanks int
	inBlockComment := false
	var activeBlockEnd string

	for scanner.Scan() {
		rawLine := scanner.Text()
		trimmed := strings.TrimSpace(rawLine)
		lines++

		if trimmed == "" {
			blanks++
			continue
		}

		isCommentLine := false

		if inBlockComment {
			comments++
			if strings.Contains(trimmed, activeBlockEnd) {
				inBlockComment = false
			}
			continue
		}

		// Cek semua aturan komentar yang berlaku untuk ekstensi ini
		for _, rule := range rules {
			if rule.IsBlock {
				if strings.HasPrefix(trimmed, rule.BlockStart) {
					isCommentLine = true
					comments++
					if !strings.Contains(trimmed[len(rule.BlockStart):], rule.BlockEnd) {
						inBlockComment = true
						activeBlockEnd = rule.BlockEnd
					}
					break
				}
			} else {
				if rule.Single != "" && strings.HasPrefix(trimmed, rule.Single) {
					isCommentLine = true
					comments++
					break
				}
			}
		}

		if isCommentLine {
			continue
		}

		code++
	}

	return FileStats{
		Path:     path,
		Lines:    lines,
		Code:     code,
		Comments: comments,
		Blanks:   blanks,
	}, nil
}
